## 功能一：PDF 转 WORD

### 1.1 需求分析

**用户场景**：
- 将 PDF 文档转换为可编辑的 Word 文档
- 保留原有排版、表格、图片
- 支持批量转换

**技术挑战**：
- PDF 是"打印格式"，不包含结构化信息
- 表格识别和重建
- 图片位置还原
- 中文字体处理

### 1.2 技术方案对比

| 方案 | 依赖 | 优点 | 缺点 | 推荐度 |
|------|------|------|------|--------|
| **pdf2docx** | pdf2docx | 专业级转换、表格支持好、中文友好 | 新增依赖 (~5MB) | ⭐⭐⭐⭐⭐ |
| PyMuPDF + python-docx | 现有 | 无新依赖 | 需自己实现布局算法、工作量大 | ⭐⭐ |
| LibreOffice CLI | 外部工具 | 转换质量高 | 需要安装 LibreOffice | ⭐⭐⭐ |
| Pandoc | 外部工具 | 已集成 | PDF转DOCX质量一般 | ⭐⭐ |

**推荐方案**：`pdf2docx` 库

# core/features/pdf/pdf_to_word.py
from PyQt6.QtCore import QObject, pyqtSignal
import os

class PDFToWordConverter(QObject):
    """
    PDF 转 WORD 转换器
    基于 pdf2docx 库实现
    """
    
    progress = pyqtSignal(int, int, str)  # current, total, message
    log = pyqtSignal(str)
    finished = pyqtSignal(int, int)  # success, fail
    
    def __init__(self, parent=None):
        super().__init__(parent)
        self._is_running = False
        self._check_dependency()
    
    def _check_dependency(self):
        """检查 pdf2docx 是否可用"""
        try:
            from pdf2docx import Converter
            self._available = True
        except ImportError:
            self._available = False
    
    def is_available(self) -> bool:
        return self._available
    
    def stop(self):
        self._is_running = False
    
    def convert_batch(
        self, 
        files: list, 
        output_dir: str = None,
        start_page: int = 0,
        end_page: int = None
    ):
        """
        批量转换 PDF 到 DOCX
        
        :param files: PDF 文件路径列表
        :param output_dir: 输出目录，None 则与源文件同目录
        :param start_page: 起始页码 (0-indexed)
        :param end_page: 结束页码，None 表示全部
        """
        if not self._available:
            self.log.emit("错误: 需要安装 pdf2docx 库")
            self.log.emit("请运行: pip install pdf2docx")
            self.finished.emit(0, len(files))
            return
        
        from pdf2docx import Converter
        
        self._is_running = True
        success = 0
        fail = 0
        total = len(files)
        
        self.log.emit(f"开始 PDF → WORD 转换，共 {total} 个文件")
        
        for i, pdf_path in enumerate(files):
            if not self._is_running:
                self.log.emit("任务已取消")
                break
            
            file_name = os.path.basename(pdf_path)
            self.progress.emit(i, total, f"正在转换: {file_name}")
            
            try:
                # 确定输出路径
                if output_dir:
                    docx_path = os.path.join(
                        output_dir, 
                        os.path.splitext(file_name)[0] + ".docx"
                    )
                else:
                    docx_path = os.path.splitext(pdf_path)[0] + ".docx"
                
                # 确保输出目录存在
                os.makedirs(os.path.dirname(docx_path) or '.', exist_ok=True)
                
                # 执行转换
                cv = Converter(pdf_path)
                cv.convert(docx_path, start=start_page, end=end_page)
                cv.close()
                
                success += 1
                self.log.emit(f"✅ 成功: {file_name}")
                
            except Exception as e:
                fail += 1
                self.log.emit(f"❌ 失败 {file_name}: {str(e)}")
        
        self.progress.emit(total, total, "完成")
        self.finished.emit(success, fail)
        self._is_running = False
    
    def convert_single(
        self, 
        pdf_path: str, 
        docx_path: str,
        start_page: int = 0,
        end_page: int = None
    ) -> bool:
        """
        转换单个 PDF 文件
        """
        if not self._available:
            return False
        
        try:
            from pdf2docx import Converter
            cv = Converter(pdf_path)
            cv.convert(docx_path, start=start_page, end=end_page)
            cv.close()
            return True
        except Exception as e:
            self.log.emit(f"转换失败: {str(e)}")
            return False
```

        self.worker_thread.started.connect(

        self.converter.stop()
        self.stop_btn.setEnabled(False)
    
    def _update_progress(self, current, total, message):
        self.progress_bar.setValue(int(current / total * 100) if total > 0 else 0)
    
    def _append_log(self, message):
        self.log_text.append(message)
    
    def _on_finished(self, success, fail):
        self.progress_bar.setValue(100)
        self.convert_btn.setEnabled(True)
        self.stop_btn.setEnabled(False)
        self._append_log(f"\n转换完成: 成功 {success} 个, 失败 {fail} 个")
        
        if self.worker_thread:
            self.worker_thread.quit()
            self.worker_thread.wait()

## 功能二：合并 WORD 文档

### 2.1 需求分析

**用户场景**：
- 将多个 Word 文档合并为一个
- 保持各文档的格式
- 可选择是否在文档间插入分页符

**技术要点**：
- 使用 python-docx 库（已有依赖）
- 处理样式冲突
- 保留页眉页脚（可选）

### 2.2 技术实现

#### 核心引擎

```python
# core/features/office/word_merger.py
from PyQt6.QtCore import QObject, pyqtSignal
from docx import Document
from docx.shared import Pt
from docx.oxml.ns import qn
from docx.oxml import OxmlElement
import os

class WordMerger(QObject):
    """
    WORD 文档合并器
    基于 python-docx 实现
    """
    
    progress = pyqtSignal(int, int, str)
    log = pyqtSignal(str)
    finished = pyqtSignal(bool, str)  # success, message/path
    
    def __init__(self, parent=None):
        super().__init__(parent)
    
    def merge(
        self, 
        files: list, 
        output_path: str,
        add_page_break: bool = True,
        keep_styles: bool = True
    ):
        """
        合并多个 DOCX 文件
        
        :param files: DOCX 文件路径列表
        :param output_path: 输出文件路径
        :param add_page_break: 是否在文档间添加分页符
        :param keep_styles: 是否尝试保留原始样式
        """
        if not files:
            self.finished.emit(False, "文件列表为空")
            return
        
        if len(files) < 2:
            self.finished.emit(False, "至少需要两个文件才能合并")
            return
        
        total = len(files)
        self.log.emit(f"开始合并 {total} 个 Word 文档")
        
        try:
            # 以第一个文档为基础
            first_file = files[0]
            self.log.emit(f"读取基础文档: {os.path.basename(first_file)}")
            self.progress.emit(0, total, f"读取: {os.path.basename(first_file)}")
            
            merged_doc = Document(first_file)
            
            # 依次追加其他文档
            for i, file_path in enumerate(files[1:], start=1):
                file_name = os.path.basename(file_path)
                self.log.emit(f"正在合并 [{i+1}/{total}]: {file_name}")
                self.progress.emit(i, total, f"合并: {file_name}")
                
                try:
                    # 添加分页符
                    if add_page_break:
                        merged_doc.add_page_break()
                    
                    # 读取要合并的文档
                    sub_doc = Document(file_path)
                    
                    # 复制段落
                    for para in sub_doc.paragraphs:
                        self._copy_paragraph(merged_doc, para)
                    
                    # 复制表格
                    for table in sub_doc.tables:
                        self._copy_table(merged_doc, table)
                        
                except Exception as e:
                    self.log.emit(f"⚠️ 合并 {file_name} 时出错: {str(e)}")
                    # 继续处理其他文件
            
            # 保存合并后的文档
            self.log.emit("正在保存合并文档...")
            self.progress.emit(total, total, "保存中...")
            
            # 确保输出目录存在
            os.makedirs(os.path.dirname(output_path) or '.', exist_ok=True)
            merged_doc.save(output_path)
            
            self.log.emit(f"✅ 合并成功: {output_path}")
            self.finished.emit(True, output_path)
            
        except Exception as e:
            self.log.emit(f"❌ 合并失败: {str(e)}")
            self.finished.emit(False, str(e))
    
    def _copy_paragraph(self, target_doc: Document, source_para):
        """复制段落到目标文档"""
        new_para = target_doc.add_paragraph()
        
        # 复制段落格式
        new_para.style = source_para.style
        new_para.paragraph_format.alignment = source_para.paragraph_format.alignment
        new_para.paragraph_format.first_line_indent = source_para.paragraph_format.first_line_indent
        new_para.paragraph_format.left_indent = source_para.paragraph_format.left_indent
        new_para.paragraph_format.right_indent = source_para.paragraph_format.right_indent
        new_para.paragraph_format.space_before = source_para.paragraph_format.space_before
        new_para.paragraph_format.space_after = source_para.paragraph_format.space_after
        new_para.paragraph_format.line_spacing = source_para.paragraph_format.line_spacing
        
        # 复制 runs (带格式的文本块)
        for run in source_para.runs:
            new_run = new_para.add_run(run.text)
            # 复制字体格式
            new_run.bold = run.bold
            new_run.italic = run.italic
            new_run.underline = run.underline
            new_run.font.size = run.font.size
            new_run.font.name = run.font.name
            new_run.font.color.rgb = run.font.color.rgb
            # 中文字体
            if run.font.name:
                new_run._element.rPr.rFonts.set(qn('w:eastAsia'), run.font.name)
    
    def _copy_table(self, target_doc: Document, source_table):
        """复制表格到目标文档"""
        # 获取源表格的行列数
        rows = len(source_table.rows)
        cols = len(source_table.columns)
        
        # 创建新表格
        new_table = target_doc.add_table(rows=rows, cols=cols)
        new_table.style = source_table.style
        
        # 复制单元格内容
        for i, row in enumerate(source_table.rows):
            for j, cell in enumerate(row.cells):
                new_cell = new_table.rows[i].cells[j]
                # 复制段落
                for para in cell.paragraphs:
                    if new_cell.paragraphs:
                        new_para = new_cell.paragraphs[0]
                        new_para.clear()
                    else:
                        new_para = new_cell.add_paragraph()
                    
                    for run in para.runs:
                        new_run = new_para.add_run(run.text)
                        new_run.bold = run.bold
                        new_run.italic = run.italic

# core/features/image/image_compressor.py
from PyQt6.QtCore import QObject, pyqtSignal
from PIL import Image
import os

class ImageCompressor(QObject):
    """
    图片压缩器
    基于 Pillow 实现
    """
    
    progress = pyqtSignal(int, int, str)
    log = pyqtSignal(str)
    finished = pyqtSignal(int, int, int)  # success, fail, saved_bytes
    
    # 压缩预设
    PRESETS = {
        'web': {'quality': 75, 'max_size': 1920, 'format': 'JPEG'},
        'email': {'quality': 60, 'max_size': 1280, 'format': 'JPEG'},
        'thumbnail': {'quality': 70, 'max_size': 300, 'format': 'JPEG'},
        'high_quality': {'quality': 90, 'max_size': None, 'format': None},
        'custom': {'quality': 80, 'max_size': None, 'format': None},
    }
    
    def __init__(self, parent=None):
        super().__init__(parent)
        self._is_running = False
    
    def stop(self):
        self._is_running = False
    
    def compress_batch(
        self,
        files: list,
        output_dir: str = None,
        quality: int = 80,
        max_width: int = None,
        max_height: int = None,
        output_format: str = None,
        keep_aspect_ratio: bool = True
    ):
        """
        批量压缩图片
        
        :param files: 图片文件路径列表
        :param output_dir: 输出目录，None 则覆盖原文件
        :param quality: JPEG/WebP 质量 (1-100)
        :param max_width: 最大宽度，None 不限制
        :param max_height: 最大高度，None 不限制
        :param output_format: 输出格式 (JPEG/PNG/WEBP)，None 保持原格式
        :param keep_aspect_ratio: 缩放时保持宽高比
        """
        self._is_running = True
        success = 0
        fail = 0
        total_saved = 0
        total = len(files)
        
        self.log.emit(f"<think>
用户要求我从截断处继续输出。上次输出结束在：

```python
        self.log.emit(f"
```

我需要继续完成这个文件的内容，从这个位置继续。
</think>
开始压缩 {total} 张图片")
        self.log.emit(f"质量: {quality}%, 最大尺寸: {max_width}x{max_height}")
        
        for i, file_path in enumerate(files):
            if not self._is_running:
                self.log.emit("任务已取消")
                break
            
            file_name = os.path.basename(file_path)
            self.progress.emit(i, total, f"压缩: {file_name}")
            
            try:
                original_size = os.path.getsize(file_path)
                
                # 确定输出路径
                if output_dir:
                    out_name = file_name
                    if output_format:
                        out_name = os.path.splitext(file_name)[0] + '.' + output_format.lower()
                    out_path = os.path.join(output_dir, out_name)
                else:
                    out_path = file_path  # 覆盖原文件
                
                # 执行压缩
                saved = self._compress_single(
                    file_path, out_path, quality, 
                    max_width, max_height, output_format, keep_aspect_ratio
                )
                
                if saved is not None:
                    success += 1
                    total_saved += saved
                    ratio = (saved / original_size * 100) if original_size > 0 else 0
                    self.log.emit(f"✅ {file_name}: 节省 {self._format_size(saved)} ({ratio:.1f}%)")
                else:
                    fail += 1
                    
            except Exception as e:
                fail += 1
                self.log.emit(f"❌ 失败 {file_name}: {str(e)}")
        
        self.progress.emit(total, total, "完成")
        self.log.emit(f"\n压缩完成: 成功 {success}, 失败 {fail}, 共节省 {self._format_size(total_saved)}")
        self.finished.emit(success, fail, total_saved)
        self._is_running = False
    
    def _compress_single(
        self,
        input_path: str,
        output_path: str,
        quality: int,
        max_width: int,
        max_height: int,
        output_format: str,
        keep_aspect_ratio: bool
    ) -> int:
        """
        压缩单张图片，返回节省的字节数
        """
        original_size = os.path.getsize(input_path)
        
        with Image.open(input_path) as img:
            # 获取原始格式
            original_format = img.format or 'JPEG'
            target_format = output_format or original_format
            
            # 调整尺寸
            if max_width or max_height:
                img = self._resize_image(img, max_width, max_height, keep_aspect_ratio)
            
            # 处理颜色模式
            if target_format.upper() in ['JPEG', 'JPG']:
                if img.mode in ('RGBA', 'LA', 'P'):
                    background = Image.new('RGB', img.size, (255, 255, 255))
                    if img.mode == 'P':
                        img = img.convert('RGBA')
                    if img.mode == 'RGBA':
                        background.paste(img, mask=img.split()[3])
                    else:
                        background.paste(img)
                    img = background
                elif img.mode != 'RGB':
                    img = img.convert('RGB')
            
            # 确保输出目录存在
            os.makedirs(os.path.dirname(output_path) or '.', exist_ok=True)
            
            # 保存参数
            save_kwargs = {}
            if target_format.upper() in ['JPEG', 'JPG']:
                save_kwargs['quality'] = quality
                save_kwargs['optimize'] = True
            elif target_format.upper() == 'PNG':
                save_kwargs['optimize'] = True
            elif target_format.upper() == 'WEBP':
                save_kwargs['quality'] = quality
                save_kwargs['method'] = 6
            
            img.save(output_path, format=target_format, **save_kwargs)
        
        new_size = os.path.getsize(output_path)
        return original_size - new_size
    
    def _resize_image(
        self, 
        img: Image.Image, 
        max_width: int, 
        max_height: int,
        keep_ratio: bool
    ) -> Image.Image:
        """调整图片尺寸"""
        orig_width, orig_height = img.size
        
        if not max_width:
            max_width = orig_width
        if not max_height:
            max_height = orig_height
        
        if orig_width <= max_width and orig_height <= max_height:
            return img  # 不需要缩放
        
        if keep_ratio:
            ratio = min(max_width / orig_width, max_height / orig_height)
            new_size = (int(orig_width * ratio), int(orig_height * ratio))
        else:
            new_size = (max_width, max_height)
        
        return img.resize(new_size, Image.Resampling.LANCZOS)
    
    def _format_size(self, size_bytes: int) -> str:
        """格式化文件大小"""
        if size_bytes < 1024:
            return f"{size_bytes} B"
        elif size_bytes < 1024 * 1024:
            return f"{size_bytes / 1024:.1f} KB"
        else:
            return f"{size_bytes / (1024 * 1024):.1f} MB"
